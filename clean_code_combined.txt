package com.battlesimulator.sortingAlgorithms;

import java.util.ArrayList;
import java.util.List;

public abstract class AbstractSortingAlgorithm <T extends Comparable<T>> {
    private String name;
    private long comparisons;
    private long executionTime;
    private long swaps;
    private List<SortStep> steps;

    public AbstractSortingAlgorithm(String name){
        this.name = name;
        this.steps = new ArrayList<>();
    }

    protected void swap(T[] arr, int i, int j) {
       T temp;
       temp = arr[i];
       arr[i] = arr[j];
       arr[j] = temp; 
       swaps++;
       steps.add(new SortStep(StepType.SWAP, i, j));
    }

    // הוספנו: פונקציה ייעודית לדריסת תא במערך (עבור Merge, Radix, Bucket)
    protected void overwrite(T[] arr, int index, T value) {
        arr[index] = value;
        swaps++; // אנחנו סופרים פעולת כתיבה/דריסה בתור "swap" כדי שזה יופיע בסטטיסטיקה במסך
        steps.add(new SortStep(StepType.OVERWRITE, index, (Integer) value));
    }

    protected int compare(T[] array, int i, int j){
        comparisons++;
        steps.add(new SortStep(StepType.COMPARISON, i, j));
        return array[i].compareTo(array[j]);
    }

    public abstract void performSort(T[] array);

    public void sort(T[] array){
        swaps = 0;
        comparisons = 0;
        steps.clear();
        
        long startTime = System.nanoTime();
        performSort(array);
        long endTime = System.nanoTime();
        executionTime = endTime - startTime;
    }

    public long getComparisons() {
        return comparisons;
    }

    public long getExecutionTime() {
        return executionTime;
    }

    public String getName() {
        return name;
    }

    public List<SortStep> getSteps() {
        return steps;
    }

    public long getSwaps() {
        return swaps;
    }
}

package com.battlesimulator.sortingAlgorithms;

public class BubbleSort<T extends Comparable<T>> extends AbstractSortingAlgorithm<T> {

    public BubbleSort() {
        super("Bubble Sort");
    }

    @Override
    public void performSort(T[] array) {
        int n = array.length;
        // We only need to do n-1 passes
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n - 1 - i; j++) {
                // Must check > 0, not == 1
                if (compare(array, j, j + 1) > 0) {
                    swap(array, j, j + 1);
                }
            }
        }
    }
}

package com.battlesimulator.sortingAlgorithms;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class BucketSort<T extends Comparable<T>> extends AbstractSortingAlgorithm<T> {

    public BucketSort() {
        super("Bucket sort");
    }

    @Override
    public void performSort(T[] array) {
        int n = array.length;
        if (n <= 1) return;

        // מציאת הערך המקסימלי כדי לדעת איך לחלק לדליים
        int max = ((Number) array[0]).intValue();
        for (int i = 1; i < n; i++) {
            int val = ((Number) array[i]).intValue();
            if (val > max) max = val;
        }

        // ניצור דליים (בדרך כלל כשורש של כמות האיברים)
        int bucketCount = (int) Math.sqrt(n);
        if (bucketCount < 1) bucketCount = 1;
        
        List<List<T>> buckets = new ArrayList<>(bucketCount);
        for (int i = 0; i < bucketCount; i++) {
            buckets.add(new ArrayList<>());
        }

        // חלוקת האיברים מהמערך לתוך הדליים המתאימים
        for (int i = 0; i < n; i++) {
            int val = ((Number) array[i]).intValue();
            int bucketIndex = (val * bucketCount) / (max + 1);
            buckets.get(bucketIndex).add(array[i]);
        }

        // מיון של כל דלי בנפרד והחזרתו למערך המקורי
        int writeIdx = 0;
        for (int i = 0; i < bucketCount; i++) {
            List<T> bucket = buckets.get(i);
            
            // ממיינים את הדלי הספציפי (ברמה הפנימית)
            Collections.sort(bucket); 
            
            // החלק החשוב שלנו: שופכים את התוכן חזרה למערך בעזרת דריסה (OVERWRITE)
            for (T targetVal : bucket) {
                
                overwrite(array, writeIdx, targetVal); // אין שגיאות steps!
                writeIdx++;
                
            }
        }
    }
}

package com.battlesimulator.sortingAlgorithms;

public class HeapSort<T extends Comparable<T>> extends AbstractSortingAlgorithm<T> {

    public HeapSort() {
        super("Heap sort");
    }

    public void performSort(T[] array) {
        int n = array.length;

        // שלב 1: בניית הערימה (Max Heap)
        // מתחילים מהאבא האחרון (n/2 - 1) ועולים למעלה עד השורש באינדקס 0
        for (int i = n / 2 - 1; i >= 0; i--) {
            heapify(array, n, i);
        }

        // שלב 2: שליפה ומיון
        // רצים מהסוף להתחלה - בכל פעם לוקחים את הכי גדול (שבראש) וזורקים לסוף
        for (int i = n - 1; i > 0; i--) {
            swap(array, 0, i); // העברת המקסימום הנוכחי לסוף המערך
            
            // תיקון הערימה עבור שאר האיברים שנשארו (הגודל מתכווץ ל-i)
            heapify(array, i, 0); 
        }
    }

    // פונקציית העזר: מחלחלת איבר קטן למטה כדי לשמור על חוקיות הערימה
    private void heapify(T[] array, int n, int i) {
        int largest = i; // מניחים כרגע שהאבא הוא הכי גדול
        int left = 2 * i + 1; // אינדקס הילד השמאלי
        int right = 2 * i + 2; // אינדקס הילד הימני

        if (left < n && compare(array, left, largest) > 0) {
            largest = left;
        }

        if (right < n && compare(array, right, largest) > 0) {
            largest = right;
        }

        if (largest != i) {
            swap(array, i, largest);
            heapify(array, n, largest);
        }
    }
}

package com.battlesimulator.sortingAlgorithms;

import java.util.ArrayList;
import java.util.List;

public class MergeSort<T extends Comparable<T>> extends AbstractSortingAlgorithm<T> {

    public MergeSort() {
        super("Merge sort");
    }

    @Override
    public void performSort(T[] array) {
        if (array == null || array.length <= 1) return;
        mergeSort(array, 0, array.length - 1);
    }

    private void mergeSort(T[] array, int left, int right) {
        if (left < right) {
            int mid = left + (right - left) / 2;
            mergeSort(array, left, mid);
            mergeSort(array, mid + 1, right);
            mergeLinear(array, left, mid, right);
        }
    }

    private void mergeLinear(T[] array, int left, int mid, int right) {
        // איחוד ליניארי קלאסי לתוך רשימת עזר
        List<T> merged = new ArrayList<>(right - left + 1);
        int i = left;
        int j = mid + 1;

        while (i <= mid && j <= right) {
            if (compare(array, i, j) <= 0) {
                merged.add(array[i]);
                i++;
            } else {
                merged.add(array[j]);
                j++;
            }
        }

        while (i <= mid) {
            merged.add(array[i]);
            i++;
        }

        while (j <= right) {
            merged.add(array[j]);
            j++;
        }

        // השינוי שלנו: במקום לחפש איפה האיבר נמצא במערך ולהחליף, 
        // פשוט עוברים על רשימת העזר ודורסים את המערך המקורי!
        for (int k = 0; k < merged.size(); k++) {
            int writeIdx = left + k;
            T targetVal = merged.get(k);
            
            // שימוש בפונקציה החדשה שעושה הכל: מעדכנת, סופרת ומתעדת אנימציה של OVERWRITE
            overwrite(array, writeIdx, targetVal); 
        }
    }
}

package com.battlesimulator.sortingAlgorithms;

import java.util.Random;

public class QuickSort<T extends Comparable<T>> extends AbstractSortingAlgorithm<T> {
    
    private Random rand;
    
    public QuickSort() {
        super("Quick Sort");
        this.rand = new Random();
    }

    @Override
    public void performSort(T[] array) {
        quickSort(array, 0, array.length - 1);
    }

    private void quickSort(T[] array, int low, int high) {
        if (low < high) {
            int pivotIndex = partition(array, low, high);

            quickSort(array, low, pivotIndex - 1);
            quickSort(array, pivotIndex + 1, high);
        }
    }
    
    private int partition(T[] array, int low, int high) {
        //Pick a random index between low and high (inclusive)
        int randomPivotIndex = rand.nextInt(high - low + 1) + low;
        
        //Swap the random element with the last element (Lomuto partition).
        swap(array, randomPivotIndex, high);

        //Standard Lomuto partitioning
        int i = low - 1;

        for (int j = low; j < high; j++) {
            if (compare(array, j, high) <= 0) {
                i++;
                swap(array, i, j);
            }
        }
        //return the pivot to his place and return the pivot position
        swap(array, i + 1, high);
        return i + 1;
    }
}

package com.battlesimulator.sortingAlgorithms;

import java.util.ArrayList;
import java.util.List;

public class RadixSort<T extends Comparable<T>> extends AbstractSortingAlgorithm<T> {

    public RadixSort() {
        super("Radix sort");
    }

    @Override
    public void performSort(T[] array) {
        int n = array.length;
        if (n <= 1) return;

        int max = ((Number) array[0]).intValue();
        for (int i = 1; i < n; i++) {
            int val = ((Number) array[i]).intValue();
            if (val > max) max = val;
        }

        // מעבר על כל ספרה (אחדות, עשרות...)
        for (int exp = 1; max / exp > 0; exp *= 10) {
            
            // חלוקה ל-10 דליים
            List<List<T>> buckets = new ArrayList<>(10);
            for (int d = 0; d < 10; d++) {
                buckets.add(new ArrayList<>());
            }

            // הכנסת האיברים לדליים לפי הספרה הנוכחית
            for (int i = 0; i < n; i++) {
                int val = ((Number) array[i]).intValue();
                int digit = (val / exp) % 10;
                buckets.get(digit).add(array[i]);
            }

            // החזרת האיברים למערך - דריסה חלקה וישירה
            int writeIdx = 0;
            for (int d = 0; d < 10; d++) {
                for (T targetVal : buckets.get(d)) {
                    
                    // כאן אנחנו קוראים לפונקציה החדשה שיצרת ב-AbstractSortingAlgorithm
                    // היא מעדכנת את המערך, סופרת את הפעולה ומתעדת את צעד האנימציה (OVERWRITE)
                    overwrite(array, writeIdx, targetVal);
                    
                    writeIdx++;
                }
            }
        }
    }
}

package com.battlesimulator.sortingAlgorithms;

public class ShellSort<T extends Comparable<T>> extends AbstractSortingAlgorithm<T> {
    
    public ShellSort() {
        super("Shell sort");
    }

    @Override
    public void performSort(T[] array) {
        int n = array.length;

        // Start with a big gap, then reduce the gap by half each iteration
        for (int gap = n / 2; gap > 0; gap /= 2) {
            
            // Perform a gapped insertion sort for this gap size
            for (int i = gap; i < n; i++) {
                int j = i;
                
                // Compare elements that are 'gap' distance apart and swap if needed
                while (j >= gap && compare(array, j - gap, j) > 0) {
                    swap(array, j - gap, j);
                    j -= gap;
                }
            }
        }
    }
}

package com.battlesimulator.sortingAlgorithms;

public class SortStep {
    StepType type;
    int indexi;
    int indexj;
    Integer value; // הוספנו: שמירת הערך לדריסה

    // בנאי מקורי להשוואות והחלפות (SWAP/COMPARISON)
    public SortStep(StepType type, int i, int j) {
        this.type = type;
        this.indexi = i;
        this.indexj = j;
    }

    // הוספנו: בנאי חדש ייעודי לדריסות (OVERWRITE)
    public SortStep(StepType type, int i, Integer value) {
        this.type = type;
        this.indexi = i;
        this.indexj = -1; // לא רלוונטי בדריסה
        this.value = value;
    }

    public StepType getType() {
        return type;
    }

    public int getIndexi() {
        return indexi;
    }

    public int getIndexj() {
        return indexj;
    }

    // הוספנו: גישה לערך החדש
    public Integer getValue() {
        return value;
    }
}

// frontend/src/App.jsx
import { useState, useEffect, useCallback } from "react";
import Visualizer from "./Visualizer";
import "./App.css";

const apiMap = {
  "Bubble Sort": "bubble",
  "Selection Sort": "selection",
  "Insertion Sort": "insertion",
  "Heap Sort": "heap",
  "Quick Sort": "quick",
  "Shell Sort": "shell",
  "Merge Sort": "merge",
  "Bucket Sort": "bucket",
  "Radix Sort": "radix",
};

const algorithms = [
  "Bubble Sort",
  "Selection Sort",
  "Insertion Sort",
  "Heap Sort",
  "Quick Sort",
  "Shell Sort",
  "Merge Sort",
  "Bucket Sort",
  "Radix Sort",
];

function App() {
  const [array, setArray] = useState([]);
  const [arraySize, setArraySize] = useState(20);

  const [fighter1, setFighter1] = useState(null);
  const [fighter2, setFighter2] = useState(null);

  const [fighter1Data, setFighter1Data] = useState(null);
  const [fighter2Data, setFighter2Data] = useState(null);

  const [activeRunners, setActiveRunners] = useState(0);
  const isRunning = activeRunners > 0;

  // פונקציה שמייצרת מערך חדש (מקבלת גודל אופציונלי כדי לתמוך בעזיבת הסליידר)
  const generateArray = (customSize = arraySize) => {
    if (isRunning) return;
    setActiveRunners(0);
    setFighter1Data(null);
    setFighter2Data(null);

    const newArray = [];
    for (let i = 0; i < customSize; i++) {
      newArray.push(Math.floor(Math.random() * 96) + 5);
    }
    setArray(newArray);
  };

  // רץ פעם אחת בלבד בטעינת האפליקציה
  useEffect(() => {
    generateArray(arraySize);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // <-- שינוי 1: בזמן הגרירה מעדכנים רק את המספר בתצוגה (בלי לקרוא ל-generateArray)
  const handleSizeChange = (e) => {
    if (isRunning) return;
    setArraySize(Number(e.target.value));
  };

  // <-- שינוי 2: פונקציה שנקראת רק כשעוזבים את הלחיצה על הסליידר!
  const handleSliderRelease = () => {
    if (isRunning) return;
    generateArray(arraySize);
  };

  // <-- שינוי 3: פונקציית Reset שמפסיקה את הקרב ומחזירה את המערך למצבו ההתחלתי
  const handleReset = () => {
    setActiveRunners(0); // עוצר את מצב הריצה ומשחרר כפתורים
    setFighter1Data(null); // מוחק את הצעדים של לוחם 1 (מאפס את ה-Visualizer שלו)
    setFighter2Data(null); // מוחק את הצעדים של לוחם 2 (מאפס את ה-Visualizer שלו)
    setFighter1(null);
    setFighter2(null);
  };

  const toggleFighter = (algName) => {
    if (isRunning) return;
    setFighter1Data(null);
    setFighter2Data(null);
    if (fighter1 === algName) {
      setFighter1(null);
    } else if (fighter2 === algName) {
      setFighter2(null);
    } else if (!fighter1) {
      setFighter1(algName);
    } else if (!fighter2) {
      setFighter2(algName);
    }
  };

  const handleFighterFinish = useCallback(() => {
    setActiveRunners((prev) => Math.max(0, prev - 1));
  }, []);

  const handleSort = async (alg1, alg2) => {
    if (!alg1 || !alg2 || isRunning) return;

    setFighter1Data(null);
    setFighter2Data(null);

    const endpoint1 = apiMap[alg1];
    const endpoint2 = apiMap[alg2];

    try {
      const request1 = fetch(`http://localhost:8080/api/sort/${endpoint1}`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ array: array }),
      });
      const request2 = fetch(`http://localhost:8080/api/sort/${endpoint2}`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ array: array }),
      });

      const [response1, response2] = await Promise.all([request1, request2]);

      const [data1, data2] = await Promise.all([
        response1.json(),
        response2.json(),
      ]);

      setFighter1Data(data1);
      setFighter2Data(data2);
      setActiveRunners(2);
    } catch (err) {
      console.error("Error fetching sort data:", err);
      setActiveRunners(0);
    }
  };

  return (
    <div className="app-container">
      <h1>Sorting Algorithms Battle Simulator</h1>

      {/* data input section */}
      <div className="top-section data-input">
        <div className="input-controls">
          <div className="size-control">
            <label htmlFor="arraySize">Array Size: {arraySize}</label>
            {/* <-- שינוי 4: הוספת onMouseUp ו-onTouchEnd לסליידר */}
            <input
              type="range"
              id="arraySize"
              min="5"
              max="100"
              value={arraySize}
              onChange={handleSizeChange}
              onMouseUp={handleSliderRelease}
              onTouchEnd={handleSliderRelease}
              disabled={isRunning}
            />
          </div>

          <button
            onClick={() => generateArray(arraySize)}
            className="generate-btn"
            disabled={isRunning}
          >
            Generate New Array
          </button>
        </div>
      </div>

      {/* central section */}
      <div className="middle-section battle-arena">
        <Visualizer
          fighterName={fighter1}
          initialArray={array}
          steps={fighter1Data?.steps}
          executionTime={fighter1Data?.executionTime}
          onFinish={handleFighterFinish}
        />

        {/* <-- שינוי 5: עטיפת כפתורי ה-START וה-RESET יחד במרכז הזירה */}
        <div className="center-controls">
          <button
            className="start-button"
            disabled={!fighter1 || !fighter2 || isRunning}
            onClick={() => handleSort(fighter1, fighter2)}
          >
            {isRunning ? "RUNNING" : "START"}
          </button>

          <button
            className="reset-button"
            // disabled={
            //   !fighter1Data ||
            //   !fighter2Data ||
            //   !isRunning ||
            //   !fighter2 ||
            //   !fighter1
            // }
            onClick={handleReset}
          >
            RESET
          </button>
        </div>

        <Visualizer
          fighterName={fighter2}
          initialArray={array}
          steps={fighter2Data?.steps}
          executionTime={fighter2Data?.executionTime}
          onFinish={handleFighterFinish}
        />
      </div>

      {/* bottom section - algorithms selection */}
      <div className="bottom-section algorithm-selector">
        {algorithms.map((alg, index) => {
          const isSelected = fighter1 === alg || fighter2 === alg;

          return (
            <div
              key={index}
              className={`fighter-card ${isSelected ? "selected" : ""} ${
                isRunning ? "disabled-card" : ""
              }`}
              onClick={() => toggleFighter(alg)}
            >
              {alg}
            </div>
          );
        })}
      </div>
    </div>
  );
}

export default App;

function Visualizer({
  fighterName,
  initialArray,
  steps,
  executionTime,
  onFinish,
  speed = 40,
}) {
  const [displayArray, setDisplayArray] = useState([]);
  const [currentStepIdx, setCurrentStepIdx] = useState(0);
  const [activeIndices, setActiveIndices] = useState([]);
  const [stepType, setStepType] = useState(null);

  const [swaps, setSwaps] = useState(0);
  const [comparisons, setComparisons] = useState(0);

  const targetSortedArray = useMemo(() => {
    return [...initialArray].sort((a, b) => a - b);
  }, [initialArray]);

  const lastSwapStepForIndex = useMemo(() => {
    const map = {};
    if (!steps) return map;
    steps.forEach((step, stepIdx) => {
      if (step.type === "SWAP" || step.type === "OVERWRITE") {
        const i = step.indexi ?? step.index1 ?? step.i;
        const j = step.indexj ?? step.index2 ?? step.j;
        if (i !== undefined && i !== -1) map[i] = stepIdx;
        if (j !== undefined && j !== -1) map[j] = stepIdx;
      }
    });
    return map;
  }, [steps]);

  useEffect(() => {
    setDisplayArray([...initialArray]);
    setCurrentStepIdx(0);
    setActiveIndices([]);
    setStepType(null);
    setSwaps(0);
    setComparisons(0);
  }, [initialArray, fighterName, steps]);

  useEffect(() => {
    if (!steps || steps.length === 0) return;

    if (currentStepIdx >= steps.length) {
      if (stepType !== "SORTED") {
        setActiveIndices([]);
        setStepType("SORTED");
        if (onFinish) onFinish();
      }
      return;
    }

    const timer = setTimeout(() => {
      const step = steps[currentStepIdx];

      const i = step.indexi ?? step.index1 ?? step.i;
      const j = step.indexj ?? step.index2 ?? step.j;
      const val = step.value ?? step.val; // הוספנו: משיכת הערך לדריסה
      const rawType = step.type ?? step.stepType;

      const normalizedType =
        rawType === "COMPARISON" || rawType === "COMPARE"
          ? "COMPARISON"
          : rawType;

      // הוספנו: אם זו דריסה, רק אינדקס i פעיל. אחרת, i ו-j
      setActiveIndices(normalizedType === "OVERWRITE" ? [i] : [i, j]);
      setStepType(normalizedType);

      if (normalizedType === "COMPARISON") {
        setComparisons((prev) => prev + 1);
      } else if (normalizedType === "SWAP") {
        setSwaps((prev) => prev + 1);
        setDisplayArray((prevArr) => {
          const newArr = [...prevArr];
          const temp = newArr[i];
          newArr[i] = newArr[j];
          newArr[j] = temp;
          return newArr;
        });
      } else if (normalizedType === "OVERWRITE") {
        // הוספנו: טיפול באנימציה של דריסה
        setDisplayArray((prevArr) => {
          const newArr = [...prevArr];
          newArr[i] = val; // משנה רק את הגובה של העמודה
          return newArr;
        });
      }

      setCurrentStepIdx((prev) => prev + 1);
    }, speed);

    return () => clearTimeout(timer);
  }, [steps, currentStepIdx, speed, stepType, onFinish]);

  const getBarStatus = (idx) => {
    if (stepType === "SORTED") return "SORTED";
    if (activeIndices.includes(idx)) return stepType;

    if (steps && steps.length > 0 && currentStepIdx > 0) {
      const lastSwap = lastSwapStepForIndex[idx] ?? -1;
      if (
        currentStepIdx > lastSwap &&
        displayArray[idx] === targetSortedArray[idx]
      ) {
        return "SORTED";
      }
    }
    return "DEFAULT";
  };

  return (
    <div className="animation-panel">
      <h2>{fighterName || "Waiting for Fighter..."}</h2>

      <div className="animation-screen">
        {fighterName ? (
          <BarsContainer>
            {displayArray.map((value, idx) => (
              <ArrayBar key={idx} $height={value} $status={getBarStatus(idx)} />
            ))}
          </BarsContainer>
        ) : (
          <p>Select an algorithm below</p>
        )}
      </div>

      <div className="stats-bar">
        <div className="stat-box">Swap/Write: {swaps}</div>
        {executionTime !== undefined && stepType === "SORTED" && (
          <div className="stat-box">
            Time: {(executionTime / 1000000).toFixed(4)} ms
          </div>
        )}
        <div className="stat-box">Comp: {comparisons}</div>
      </div>
    </div>
  );
}

export default Visualizer;
